cmake_minimum_required(VERSION 3.10)
project(amr_rescue LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(icraft-zg330backend REQUIRED)
find_package(icraft-hostbackend REQUIRED)
if(NOT ORIGINAL_ROOT)
  message(FATAL_ERROR "Set ORIGINAL_ROOT to existing board deployment")
endif()
add_executable(npu_stream npu_stream.cpp)
target_include_directories(npu_stream PRIVATE "${ORIGINAL_ROOT}/board_package" "${ORIGINAL_ROOT}/sdr_adapter")
target_link_libraries(npu_stream PRIVATE Icraft::ZG330Backend Icraft::HostBackend pthread dl)
